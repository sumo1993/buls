// Test API
fetch('/api/event-registrations?eventId=4')
  .then(r => r.json())
  .then(data => {
    console.log('API Response:', data);
    console.log('Total attendees:', data.registrations?.length);
    
    if (data.registrations && data.registrations.length > 0) {
      // Check phone numbers
      const withPhone = data.registrations.filter(a => 
        a.phone_number && a.phone_number.trim().length > 0
      );
      console.log('With phone numbers:', withPhone.length);
      
      // Search for 7054636640
      const found = data.registrations.filter(a => 
        a.phone_number && a.phone_number.includes('7054636640')
      );
      console.log('Found with 7054636640:', found);
      
      // Show first few attendees
      console.log('First 3 attendees:', data.registrations.slice(0, 3));
    }
  });
